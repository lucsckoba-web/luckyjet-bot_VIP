async function verifyAccess(isAuto = false) { const userId = document.getElementById('user-id').value.trim(); if (!userId) { showMsg("Veuillez entrer votre ID 1win", "error"); return; } try { // 1. Vérification du statut de l'utilisateur sur le Worker const statusRes = await fetch(`${WORKER_URL}/?id=${userId}`); const statusData = await statusRes.json(); // CAS 1 : L'ID n'existe pas dans KV (non inscrit avec le code ZIKOONE) if (!statusData.registered) { showMsg("❌ ACCÈS REFUSÉ\nCet ID n'est pas rattaché au code promo ZIKOONE.\n\n1. Crée un nouveau compte avec le code ZIKOONE.\n2. Effectue ton premier dépôt.", "error"); localStorage.removeItem('luckyjet_vip_id'); return; } // CAS 2 : Inscription détectée mais premier dépôt non effectué (deposit == 0) if (statusData.deposit === 0) { showMsg("⚠️ PREMIER DÉPÔT REQUIS\nTon ID est bien rattaché au code ZIKOONE, mais ton premier dépôt n'a pas encore été détecté.\n\nRecharge ton compte puis réessaie.", "error"); localStorage.removeItem('luckyjet_vip_id'); return; } // CAS 3 : Compte VIP Validé -> Récupération des signaux const signalsRes = await fetch(`${WORKER_URL}/get-signals?id=${userId}`); const data = await signalsRes.json(); if (data.error) { showMsg("❌ " + data.error, "error"); localStorage.removeItem('luckyjet_vip_id'); return; } localStorage.setItem('luckyjet_vip_id', userId); if (!isAuto) showMsg("✅ ACCÈS VÉRIFIÉ AVEC SUCCÈS !", "success"); activeSignals = data.signals || []; currentIndex = 0; let cleanSessionName = (data.sessionName || "SESSION ACTIVE").replace(/^Ex:\s*/i, ''); document.getElementById('session-title').innerText = cleanSessionName; setTimeout(() => { document.getElementById('login-screen').classList.add('hidden'); document.getElementById('signal-screen').classList.remove('hidden'); updateSignalDisplay(); if (!autoCheckTimer) { autoCheckTimer = setInterval(updateButtonStatus, 2000); } }, isAuto ? 0 : 800); } catch (e) { showMsg("❌ Connexion au serveur impossible", "error"); } }
